# Prep UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the existing REST API (Plans 1-2) into a usable mobile-first web app: convert long-running endpoints to background tasks with a pollable stage-status endpoint, then build the static HTML/CSS/JS UI (OLED black, Apple-like, green accent — already approved via mockup) on top of it, and wire up remote access via Tailscale.

**Architecture:** Backend first (Tasks 1-5): a generic `run_with_new_store` helper lets any existing service function run as a FastAPI `BackgroundTasks` job with its own fresh SQLite connection, so `POST` endpoints return `202` immediately instead of blocking for minutes; a new `GET /jobs/{id}/stages` endpoint is the polling target the UI uses to track progress. Frontend (Tasks 6-11): no build step, no framework — plain static HTML/CSS/JS served by FastAPI's `StaticFiles`, one page per screen, a shared `app.css`/`app.js`. Tailscale (Task 12): the server already listens on `0.0.0.0`; Tailscale reachability is a run-script/documentation concern, not new application code.

**Tech Stack:** Same as Plans 1-2 (Python 3.11+, FastAPI, SQLite, pytest) plus FastAPI's built-in `BackgroundTasks` and `StaticFiles` — no new dependencies.

## Global Constraints

- Long-running `POST` endpoints (`ingest/local`, `uploads/.../complete`, `extract-frames`, `recommend`) return HTTP `202` with `{"status": "started"}` and do the real work via `BackgroundTasks.add_task(run_with_new_store, service_fn, ...)`. Each background task opens its own SQLite connection (`db.connect()`) and closes it when done — never reuses the request-scoped connection from `get_job_store`, since its lifetime relative to background tasks is not something this project relies on.
- Service functions already record failures into `job_stages` before raising (established in Plans 1-2); `run_with_new_store` swallows the exception after that point — the job's stage status via `GET /jobs/{id}/stages` is the UI's only signal, there is no synchronous error response to show for these endpoints anymore.
- Endpoints that are fast and don't need backgrounding (`POST /jobs`, `GET /jobs`, `POST /jobs/{id}/uploads`, `PUT .../chunks/{i}`, `GET .../chunks`, `GET /jobs/{id}/stages`, `GET /jobs/{id}/recommendation`, `GET /jobs/{id}/frames`, `GET /jobs/{id}/frames/{filename}`) stay synchronous, unchanged from their existing behavior.
- Frontend is plain HTML/CSS/JS, no build tooling, no framework, served via `StaticFiles(directory=..., html=True)` mounted at `/` — mounted *after* all API routers so explicit API routes always win over static file matching.
- Visual style (already approved): OLED black background (`#050506`), rounded cards, green accent `#30D158` for success/active states, orange `#FF9F0A` for warnings/failed/cloud states, gray for neutral/pending, `-apple-system` font stack, minimalist single-focus screens.
- Client-supplied filenames used in filesystem paths are always sanitized to a bare basename and rejected outright if that changes the value (established in Plan 1's security fix) — applied again here for the new `GET /jobs/{id}/frames/{filename}` endpoint.
- Frontend pages get a pytest smoke test (served, correct status, expected DOM hooks present) as the automated gate; actual visual/interactive correctness should be checked in a browser (e.g. via the `chrome-devtools` tooling if available) before a frontend task is reported done — note in the task report whether this was possible.

```python
from pydantic import BaseModel

    raw_video_path: str | None

    status: str

    finished_at: str | None

    error: str | None

    return JobResponse(**job.__dict__)


@router.get("/{job_id}", response_model=JobResponse)

    if job is None:
        raise HTTPException(status_code=404, detail="Job not found")

@router.get("", response_model=list[JobResponse])
def list_jobs(store: JobStore = Depends(get_job_store)) -> list[JobResponse]:

@router.get("/{job_id}/stages", response_model=list[JobStageResponse])

) -> list[JobStageResponse]:
    if store.get_job(job_id) is None:
        raise HTTPException(status_code=404, detail="Job not found")
    return [JobStageResponse(**stage.__dict__) for stage in store.list_stages(job_id)]
```

```python
from __future__ import annotations

from collections.abc import Callable, Iterator
from typing import Any

from app import db
from app.job_store import JobStore


def get_job_store() -> Iterator[JobStore]:
    conn = db.connect()
    try:
        yield JobStore(conn)
    finally:
        conn.close()


def run_with_new_store(func: Callable[..., Any], *args: Any, **kwargs: Any) -> None:
    conn = db.connect()
    store = JobStore(conn)
    try:
        func(store, *args, **kwargs)
    except Exception:
        # Service functions already record failures into job_stages before
        # raising; a background task has no caller to propagate the
        # exception to, so swallow it here — the job's stage status (polled
        # via GET /jobs/{id}/stages) is the UI's source of truth.
        pass
    finally:
        conn.close()
```

```python
from __future__ import annotations

from pathlib import Path

from fastapi import APIRouter, BackgroundTasks, Depends, HTTPException, Request
from pydantic import BaseModel

from app.deps import get_job_store, run_with_new_store
from app.ingestion import import_local_file
from app.job_store import JobStore
from app.uploads import UploadError, complete_upload, received_chunks, save_chunk, start_upload

router = APIRouter(prefix="/jobs", tags=["ingestion"])


class LocalImportRequest(BaseModel):
    source_path: str


class StartUploadRequest(BaseModel):
    filename: str
    total_chunks: int


@router.post("/{job_id}/ingest/local", status_code=202)
def ingest_local(
    job_id: str,
    body: LocalImportRequest,
    background_tasks: BackgroundTasks,
    store: JobStore = Depends(get_job_store),
) -> dict[str, str]:
    if store.get_job(job_id) is None:
        raise HTTPException(status_code=404, detail="Job not found")
    background_tasks.add_task(run_with_new_store, import_local_file, job_id, Path(body.source_path))
    return {"status": "started"}


@router.post("/{job_id}/uploads")
def create_upload(
    job_id: str, body: StartUploadRequest, store: JobStore = Depends(get_job_store)
) -> dict[str, str]:
    if store.get_job(job_id) is None:
        raise HTTPException(status_code=404, detail="Job not found")
    try:
        upload_id = start_upload(job_id, body.filename, body.total_chunks)
    except UploadError as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc
    return {"upload_id": upload_id}


@router.put("/{job_id}/uploads/{upload_id}/chunks/{chunk_index}")
async def upload_chunk(job_id: str, upload_id: str, chunk_index: int, request: Request) -> dict[str, int]:
    data = await request.body()
    try:
        save_chunk(job_id, upload_id, chunk_index, data)
    except UploadError as exc:
        raise HTTPException(status_code=404, detail=str(exc)) from exc
    return {"received": chunk_index}


@router.get("/{job_id}/uploads/{upload_id}/chunks")
def list_received_chunks(job_id: str, upload_id: str) -> dict[str, list[int]]:
    try:
        return {"received_chunks": received_chunks(job_id, upload_id)}
    except UploadError as exc:
        raise HTTPException(status_code=404, detail=str(exc)) from exc


@router.post("/{job_id}/uploads/{upload_id}/complete", status_code=202)
def finish_upload(
    job_id: str,
    upload_id: str,
    background_tasks: BackgroundTasks,
    store: JobStore = Depends(get_job_store),
) -> dict[str, str]:
    if store.get_job(job_id) is None:
        raise HTTPException(status_code=404, detail="Job not found")
    background_tasks.add_task(run_with_new_store, complete_upload, job_id, upload_id)
    return {"status": "started"}
```

```python
from __future__ import annotations

import json
from pathlib import Path

from fastapi import APIRouter, BackgroundTasks, Depends, HTTPException
from pydantic import BaseModel

from app.deps import get_job_store, run_with_new_store
from app.job_store import JobStore
from app.recommendation_service import compute_recommendation

router = APIRouter(prefix="/jobs", tags=["recommendation"])


class RecommendRequest(BaseModel):
    srt_path: str | None = None


@router.post("/{job_id}/recommend", status_code=202)
def recommend_endpoint(
    job_id: str,
    body: RecommendRequest,
    background_tasks: BackgroundTasks,
    store: JobStore = Depends(get_job_store),
) -> dict[str, str]:
    if store.get_job(job_id) is None:
        raise HTTPException(status_code=404, detail="Job not found")
    srt_path = Path(body.srt_path) if body.srt_path else None
    background_tasks.add_task(run_with_new_store, compute_recommendation, job_id, srt_path)
    return {"status": "started"}


@router.get("/{job_id}/recommendation")
def get_recommendation(job_id: str, store: JobStore = Depends(get_job_store)) -> dict:
    job = store.get_job(job_id)
    if job is None:
        raise HTTPException(status_code=404, detail="Job not found")
    if job.recommendation_json is None:
        raise HTTPException(status_code=404, detail="Recommendation not computed yet")
    return json.loads(job.recommendation_json)
```

```javascript
async function apiGet(path) {
  const response = await fetch(path);
  if (!response.ok) {
    throw new Error(`GET ${path} failed: ${response.status}`);
  }
  return response.json();
}

async function apiPost(path, body) {
  const response = await fetch(path, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body || {}),
  });
  if (!response.ok) {
    throw new Error(`POST ${path} failed: ${response.status}`);
  }
  return response.json();
}

function jobIdFromQuery() {
  return new URLSearchParams(window.location.search).get("id");
}

function pollStages(jobId, onUpdate, intervalMs) {
  intervalMs = intervalMs || 2000;
  let stopped = false;

  async function tick() {
    if (stopped) return;
    try {
      const stages = await apiGet(`/jobs/${jobId}/stages`);
      onUpdate(stages);
    } catch (err) {
      console.error("poll failed", err);
    }
    if (!stopped) {
      setTimeout(tick, intervalMs);
    }
  }

  tick();
  return function stop() {
    stopped = true;
  };
}
```

```html
          s.status === "done" ? "pill-green" : s.status === "failed" ? "pill-orange" : "pill-gray";
        const errorLine = s.error ? `<div class="job-meta">${s.error}</div>` : "";
        return `
          <div class="job-row">
            <div class="job-info">
              <div class="job-name">${STAGE_LABELS[s.stage] || s.stage}</div>
              ${errorLine}
            </div>
            <span class="pill ${pillClass}">${s.status}</span>
          </div>
        `;
      })
      .join("");
  }

  function stageStatus(stages, name) {
    const stage = stages.find((s) => s.stage === name);
    return stage ? stage.status : null;
  }

  function updateActions(stages) {
    const ingestionDone = stageStatus(stages, "ingestion") === "done";
    const extractionDone = stageStatus(stages, "frame_extraction") === "done";
    const recommendationDone = stageStatus(stages, "recommendation") === "done";

    document.getElementById("btn-extract").style.display =
      ingestionDone && !extractionDone ? "block" : "none";

    const framesLink = document.getElementById("link-frames");
    framesLink.style.display = extractionDone ? "block" : "none";
    framesLink.href = `/frames.html?id=${jobId}`;

    document.getElementById("btn-recommend").style.display =
      ingestionDone && !recommendationDone ? "block" : "none";

    const recommendationLink = document.getElementById("link-recommendation");
    recommendationLink.style.display = recommendationDone ? "block" : "none";
    recommendationLink.href = `/recommendation.html?id=${jobId}`;
  }

  async function refresh() {
    const job = await apiGet(`/jobs/${jobId}`);
    document.getElementById("job-name").textContent = job.name;
    document.getElementById("job-meta").textContent = new Date(job.created_at).toLocaleString("ro-RO");

    const stages = await apiGet(`/jobs/${jobId}/stages`);
    renderStages(stages);
    updateActions(stages);
  }

  document.getElementById("btn-extract").addEventListener("click", async () => {
    await apiPost(`/jobs/${jobId}/extract-frames`, {});
    refresh();
  });

  document.getElementById("btn-recommend").addEventListener("click", async () => {
    await apiPost(`/jobs/${jobId}/recommend`, {});
    refresh();
  });

  refresh();
  pollStages(jobId, (stages) => {
    renderStages(stages);
    updateActions(stages);
  });
</script>
</body>
</html>
```

```python
from __future__ import annotations

from pathlib import Path

from fastapi import APIRouter, BackgroundTasks, Depends, HTTPException
from fastapi.responses import FileResponse
from pydantic import BaseModel

from app.deps import get_job_store, run_with_new_store
from app.extraction import extract_frames
from app.job_store import JobStore
from app.paths import frames_dir

router = APIRouter(prefix="/jobs", tags=["frames"])


class ExtractFramesRequest(BaseModel):
    selection_method: str = "best-n"
    num_frames: int = 300
    fps: int = 10


@router.post("/{job_id}/extract-frames", status_code=202)
def extract_frames_endpoint(
    job_id: str,
    body: ExtractFramesRequest,
    background_tasks: BackgroundTasks,
    store: JobStore = Depends(get_job_store),
) -> dict[str, str]:
    if store.get_job(job_id) is None:
        raise HTTPException(status_code=404, detail="Job not found")
    background_tasks.add_task(
        run_with_new_store,
        extract_frames,
        job_id,
        selection_method=body.selection_method,
        num_frames=body.num_frames,
        fps=body.fps,
    )
    return {"status": "started"}


@router.get("/{job_id}/frames")
def list_frames(job_id: str, store: JobStore = Depends(get_job_store)) -> dict[str, list[str]]:
    job = store.get_job(job_id)
    if job is None:
        raise HTTPException(status_code=404, detail="Job not found")
    if job.frames_dir is None:
        return {"frames": []}
    directory = frames_dir(job_id)
    names = sorted(p.name for p in directory.glob("*") if p.is_file())
    return {"frames": names}


@router.get("/{job_id}/frames/{filename}")
def get_frame(
    job_id: str, filename: str, store: JobStore = Depends(get_job_store)
) -> FileResponse:
    if store.get_job(job_id) is None:
        raise HTTPException(status_code=404, detail="Job not found")
    safe_name = Path(filename).name
    if not safe_name or safe_name in (".", "..") or safe_name != filename:
        raise HTTPException(status_code=400, detail=f"Invalid filename: {filename!r}")
    frame_path = frames_dir(job_id) / safe_name
    if not frame_path.is_file():
        raise HTTPException(status_code=404, detail="Frame not found")
    return FileResponse(frame_path)
```
